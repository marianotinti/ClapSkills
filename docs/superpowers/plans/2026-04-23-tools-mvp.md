# Tools MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a new top-level `Tools` product area, separate from `Skills`, where users can generate, preview, edit, save, duplicate, export, import, and reopen reusable React tools while preserving the current ClapSkills visual style.

**Architecture:** Add a feature-scoped `src/features/tools/` domain with its own types, storage, context, pages, UI components, and server helpers. Keep `Skills` untouched, add a new `/tools` route + navbar entry, use Sandpack for isolated live preview/editing, and add a dedicated `/api/tools/generate` Express endpoint that returns validated JSON instead of n8n workflow metadata.

**Tech Stack:** React 19, TypeScript, Vite, Express, react-router-dom, Tailwind v4, `@codesandbox/sandpack-react`, Vitest, React Testing Library, `@testing-library/user-event`, jsdom

Create `src/features/tools/lib/storage.ts`:

```ts
import { DEFAULT_TOOL_FILES } from './defaultToolFiles';
import { ToolRecord } from '../types';

const STORAGE_KEY = 'clapskills.tools';

function nowIso(): string {
  return new Date().toISOString();
}

export function createEmptyToolRecord(): ToolRecord {
  const timestamp = nowIso();

  return {
    id: crypto.randomUUID(),
    name: 'Untitled Tool',
    description: 'A reusable React tool for ClapSkills.',
    prompt: '',
    files: DEFAULT_TOOL_FILES,
    entryFile: '/App.tsx',
    tags: ['react-tool'],
    createdAt: timestamp,
    updatedAt: timestamp,
  };
}

export function loadStoredTools(): ToolRecord[] {
  const raw = localStorage.getItem(STORAGE_KEY);

  if (!raw) return [];

  try {
    const parsed = JSON.parse(raw) as ToolRecord[];
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

export function saveStoredTools(tools: ToolRecord[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(tools));
}

export function duplicateToolRecord(tool: ToolRecord): ToolRecord {
  const timestamp = nowIso();

  return {
    ...tool,
    id: crypto.randomUUID(),
    name: `${tool.name} Copy`,
    createdAt: timestamp,
    updatedAt: timestamp,
  };
}

export function exportTools(tools: ToolRecord[]): string {
  return JSON.stringify({ tools }, null, 2);
}

export function importTools(payload: string): ToolRecord[] {
  const parsed = JSON.parse(payload) as { tools?: ToolRecord[] };
  return Array.isArray(parsed.tools) ? parsed.tools : [];
}
```

Create `src/features/tools/context/ToolContext.tsx`:

```tsx
import {
  createContext,
  ReactNode,
  useContext,
  useMemo,
  useState,
} from 'react';
import { ToolRecord } from '../types';
import {
  createEmptyToolRecord,
  duplicateToolRecord,
  loadStoredTools,
  saveStoredTools,
} from '../lib/storage';

interface ToolContextValue {
  tools: ToolRecord[];
  selectedToolId: string;
  selectedTool: ToolRecord | null;
  createTool: () => ToolRecord;
  saveTool: (tool: ToolRecord) => void;
  updateTool: (toolId: string, updater: (tool: ToolRecord) => ToolRecord) => void;
  duplicateTool: (toolId: string) => void;
  selectTool: (toolId: string) => void;
  replaceAllTools: (tools: ToolRecord[]) => void;
}

const ToolContext = createContext<ToolContextValue | undefined>(undefined);

export function ToolProvider({ children }: { children: ReactNode }) {
  const [tools, setTools] = useState<ToolRecord[]>(() => loadStoredTools());
  const [selectedToolId, setSelectedToolId] = useState<string>(() => loadStoredTools()[0]?.id ?? '');

  const persist = (nextTools: ToolRecord[]) => {
    setTools(nextTools);
    saveStoredTools(nextTools);
  };

  const createTool = (): ToolRecord => {
    const nextTool = createEmptyToolRecord();
    const nextTools = [nextTool, ...tools];
    persist(nextTools);
    setSelectedToolId(nextTool.id);
    return nextTool;
  };

  const saveTool = (tool: ToolRecord) => {
    const existing = tools.some((current) => current.id === tool.id);
    const nextTools = existing
      ? tools.map((current) => (current.id === tool.id ? tool : current))
      : [tool, ...tools];

    persist(nextTools);
    setSelectedToolId(tool.id);
  };

  const updateTool = (toolId: string, updater: (tool: ToolRecord) => ToolRecord) => {
    const nextTools = tools.map((tool) => (tool.id === toolId ? updater(tool) : tool));
    persist(nextTools);
  };

  const duplicateTool = (toolId: string) => {
    const source = tools.find((tool) => tool.id === toolId);
    if (!source) return;

    const nextTool = duplicateToolRecord(source);
    persist([nextTool, ...tools]);
    setSelectedToolId(nextTool.id);
  };

  const replaceAllTools = (nextTools: ToolRecord[]) => {
    persist(nextTools);
    setSelectedToolId(nextTools[0]?.id ?? '');
  };

  const selectedTool = useMemo(
    () => tools.find((tool) => tool.id === selectedToolId) ?? null,
    [selectedToolId, tools],
  );

  return (
    <ToolContext.Provider
      value={{
        tools,
        selectedToolId,
        selectedTool,
        createTool,
        saveTool,
        updateTool,
        duplicateTool,
        selectTool: setSelectedToolId,
        replaceAllTools,
      }}
    >
      {children}
    </ToolContext.Provider>
  );
}

export function useTools(): ToolContextValue {
  const value = useContext(ToolContext);

  if (!value) {
    throw new Error('useTools must be used within a ToolProvider');
  }

  return value;
}
```

```ts
import { DEFAULT_TOOL_FILES } from '../lib/defaultToolFiles';
import { GenerateToolResponse, ToolFileMap } from '../types';

function isToolFileMap(value: unknown): value is ToolFileMap {
  return typeof value === 'object' && value !== null;
}

export function normalizeToolResponse(value: unknown): GenerateToolResponse {
  if (typeof value !== 'object' || value === null) {
    throw new Error('Model response must be an object.');
  }

  const record = value as Record<string, unknown>;
  const files = isToolFileMap(record.files) ? record.files : DEFAULT_TOOL_FILES;

  if (typeof files['/App.tsx'] !== 'string' || !files['/App.tsx'].includes('export default')) {
    throw new Error('Tool response must include /App.tsx with a default export.');
  }

  return {
    name: typeof record.name === 'string' ? record.name : 'Generated Tool',
    description:
      typeof record.description === 'string'
        ? record.description
        : 'A generated React tool for ClapSkills.',
    files,
  };
}
```

```ts
import { GenerateToolResponse } from './types';

export async function generateTool(prompt: string): Promise<GenerateToolResponse> {
  const response = await fetch('/api/tools/generate', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ prompt }),
  });

  if (!response.ok) {
    throw new Error(`Tool generation failed with status ${response.status}`);
  }

  return response.json() as Promise<GenerateToolResponse>;
}
```

```ts
import { normalizeToolResponse } from './src/features/tools/server/normalizeToolResponse.ts';

app.post('/api/tools/generate', async (req, res) => {
  try {
    const { prompt } = req.body as { prompt?: string };

    if (!prompt?.trim()) {
      return res.status(400).json({ error: 'Prompt is required.' });
    }

    const geminiKey = process.env.GEMINI_API_KEY;
    const modelName = process.env.TOOLS_MODEL_NAME || 'gemini-2.5-pro';

    if (!geminiKey) {
      return res.json(
        normalizeToolResponse({
          name: 'Mock Tool',
          description: prompt,
          files: {
            '/App.tsx': `export default function App() {
  return <div className="p-8 text-white">Mock tool for: ${prompt.replace(/`/g, '')}</div>;
}`,
          },
        }),
      );
    }

    const ai = new GoogleGenAI({ apiKey: geminiKey });
    const response: any = await ai.models.generateContent({
      model: modelName,
      contents: [{ role: 'user', parts: [{ text: prompt }] }],
      config: {
        responseMimeType: 'application/json',
        systemInstruction: `You generate reusable React MVP tools for ClapSkills.
Return JSON only with this shape:
{
  "name": "Tool name",
  "description": "Short summary",
  "files": {
    "/App.tsx": "export default function App() { ... }"
  }
}
Rules:
- Only generate /App.tsx and optional /components/*.tsx files
- No external packages
- Use React + Tailwind classes only
- /App.tsx must default export a component`,
      },
    });

    const text = response.text ?? response.candidates?.[0]?.content?.parts?.[0]?.text;
    const parsed = JSON.parse(text);

    return res.json(normalizeToolResponse(parsed));
  } catch (error) {
    return res.status(500).json({ error: String(error) });
  }
});
```

```tsx
import { useState } from 'react';
import { generateTool } from '../api';
import { useTools } from '../context/ToolContext';
import { createEmptyToolRecord, exportTools, importTools } from '../lib/storage';
import { ToolLibrarySidebar } from '../components/ToolLibrarySidebar';
import { ToolPromptPanel } from '../components/ToolPromptPanel';
import { ToolWorkspace } from '../components/ToolWorkspace';

export function ToolsPage() {
  const { replaceAllTools, saveTool, selectedTool, updateTool, tools } = useTools();
  const [prompt, setPrompt] = useState('');
  const [isGenerating, setIsGenerating] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleGenerate = async () => {
    try {
      setIsGenerating(true);
      setError(null);

      const result = await generateTool(prompt);
      const base = selectedTool ?? createEmptyToolRecord();

      saveTool({
        ...base,
        name: result.name,
        description: result.description,
        prompt,
        files: result.files,
        updatedAt: new Date().toISOString(),
      });
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : 'Failed to generate tool.');
    } finally {
      setIsGenerating(false);
    }
  };

  const handleExport = async () => {
    const blob = new Blob([exportTools(tools)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement('a');
    anchor.href = url;
    anchor.download = 'clapskills-tools.json';
    anchor.click();
    URL.revokeObjectURL(url);
  };

  const handleImport = async (file: File) => {
    const payload = await file.text();
    replaceAllTools(importTools(payload));
  };

  return (
    <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-10">
      <div className="mb-8">
        <h1 className="text-3xl font-bold text-on-surface">Tools</h1>
        <p className="text-on-surface-variant">Generate and save reusable React tools.</p>
      </div>

      <div className="grid grid-cols-1 gap-6 xl:grid-cols-[320px_minmax(0,1fr)]">
        <ToolLibrarySidebar onExport={handleExport} onImport={handleImport} />
        <div className="space-y-6">
          <ToolPromptPanel
            prompt={prompt}
            isGenerating={isGenerating}
            error={error}
            onPromptChange={setPrompt}
            onGenerate={handleGenerate}
          />
          <ToolWorkspace
            onNameChange={(name) =>
              selectedTool &&
              updateTool(selectedTool.id, (tool) => ({
                ...tool,
                name,
                updatedAt: new Date().toISOString(),
              }))
            }
            onCodeChange={(code) =>
              selectedTool &&
              updateTool(selectedTool.id, (tool) => ({
                ...tool,
                files: {
                  ...tool.files,
                  '/App.tsx': code,
                },
                updatedAt: new Date().toISOString(),
              }))
            }
          />
        </div>
      </div>
    </div>
  );
}
```

## Rollout Notes

- Keep `Tools` on its own `/tools` route and do not link it into the `Skills` execution flows for the MVP.

- Keep the sandbox intentionally constrained:
  - Required: `/App.tsx`
  - Optional: `/components/*.tsx`
  - No external packages
  - Tailwind utility classes only
- Treat the current `.env.example` credentials as sensitive during implementation review; avoid propagating existing secrets into the plan or follow-up PRs.
